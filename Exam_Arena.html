function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var quizBank = { Math: [], Science: [], GK: [] }; // Function to read individual subject sheets function parseSubjectSheet(sheetName, subjKey) { var sh = ss.getSheetByName(sheetName); if (!sh || sh.getLastRow() <= 1) return; var rows = sh.getRange(2, 1, sh.getLastRow() - 1, 8).getValues(); var optMap = { "A": 0, "B": 1, "C": 2, "D": 3, "a": 0, "b": 1, "c": 2, "d": 3, "1": 0, "2": 1, "3": 2, "4": 3 }; var grouped = {}; for (var k = 0; k < rows.length; k++) { var r = rows[k]; var lvl = Number(r[0]) || 1; var sTitle = (r[1] || ("Set " + lvl)).toString().trim(); var qText = (r[2] || "").toString().trim(); var oA = (r[3] || "").toString().trim(); var oB = (r[4] || "").toString().trim(); var oC = (r[5] || "").toString().trim(); var oD = (r[6] || "").toString().trim(); var ansLetter = (r[7] || "A").toString().trim(); var ansIdx = (optMap[ansLetter] !== undefined) ? optMap[ansLetter] : 0; if (!qText) continue; if (!grouped[lvl]) { grouped[lvl] = { level: lvl, title: sTitle, questions: [] }; } grouped[lvl].questions.push({ q: qText, options: [oA, oB, oC, oD], ans: ansIdx }); } for (var l in grouped) { quizBank[subjKey].push(grouped[l]); } quizBank[subjKey].sort(function(a, b) { return a.level - b.level; }); } parseSubjectSheet("Math_Bank", "Math"); parseSubjectSheet("Science_Bank", "Science"); parseSubjectSheet("GK_Bank", "GK"); // Leaderboard & Live Active Users from Users_DB var uSheet = ss.getSheetByName("Users_DB"); var top5 = [], activeUsers = []; if (uSheet && uSheet.getLastRow() > 1) { var uRows = uSheet.getRange(2, 1, uSheet.getLastRow() - 1, 10).getValues(); // Filter out empty rows var validUsers = []; for (var v = 0; v < uRows.length; v++) { if (uRows[v][0]) validUsers.push(uRows[v]); } // Sort by XP descending validUsers.sort(function(a, b) { return (Number(b[5]) || 0) - (Number(a[5]) || 0); }); for (var i = 0; i < Math.min(5, validUsers.length); i++) { top5.push({ rank: i + 1, name: validUsers[i][2] || validUsers[i][0], totalXP: Number(validUsers[i][5]) || 0 }); } for (var u = 0; u < Math.min(8, validUsers.length); u++) { activeUsers.push(validUsers[u][2] || validUsers[u][0]); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", quizzes: quizBank, top5: top5, activeUsers: activeUsers })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); var uSheet = ss.getSheetByName("Users_DB"); if (!uSheet) { uSheet = ss.insertSheet("Users_DB"); uSheet.appendRow(["Username", "PIN", "StudentName", "FatherName", "DOB", "TotalXP", "MathUnlocked", "ScienceUnlocked", "GKUnlocked", "LastActive"]); } var rows = uSheet.getLastRow() > 1 ? uSheet.getRange(2, 1, uSheet.getLastRow() - 1, 10).getValues() : []; // Helper function for date matching function normalizeDate(val) { if (!val) return ""; if (val instanceof Date) { var y = val.getFullYear(); var m = ("0" + (val.getMonth() + 1)).slice(-2); var d = ("0" + val.getDate()).slice(-2); return y + "-" + m + "-" + d; } return val.toString().trim(); } // 1. REGISTER NEW ACCOUNT if (data.action === "register") { var reqUser = (data.username || "").toLowerCase().trim().replace(/\s+/g, ''); for (var i = 0; i < rows.length; i++) { if (rows[i][0].toString().toLowerCase().trim() === reqUser) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "❌ यह Username पहले से मौजूद है! कृपया दूसरा यूज़रनेम चुनें।" })).setMimeType(ContentService.MimeType.JSON); } } uSheet.appendRow([ reqUser, data.pin.toString().trim(), data.studentName.trim(), data.fatherName.trim(), "'" + data.dob.toString().trim(), // Quote prevents auto-date mess 0, 1, 1, 1, new Date() ]); return ContentService.createTextOutput(JSON.stringify({ status: "success", user: { username: reqUser, studentName: data.studentName.trim(), totalXP: 0, unlockedLevels: { Math: 1, Science: 1, GK: 1 } } })).setMimeType(ContentService.MimeType.JSON); } // 2. LOGIN if (data.action === "login") { var lUser = (data.username || "").toLowerCase().trim().replace(/\s+/g, ''); var foundIdx = -1; var foundRow = null; for (var j = 0; j < rows.length; j++) { if (rows[j][0].toString().toLowerCase().trim() === lUser) { foundIdx = j + 2; foundRow = rows[j]; break; } } if (foundIdx === -1) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "❌ यह Username रजिस्टर नहीं है! कृपया स्पेलिंग जांचें या नया अकाउंट बनाएं।" })).setMimeType(ContentService.MimeType.JSON); } if (foundRow[1].toString().trim() !== data.pin.toString().trim()) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "❌ गलत 4-Digit PIN! यदि PIN भूल गए हैं तो 'Forgot PIN' से रिकवर करें।" })).setMimeType(ContentService.MimeType.JSON); } uSheet.getRange(foundIdx, 10).setValue(new Date()); return ContentService.createTextOutput(JSON.stringify({ status: "success", user: { username: foundRow[0], studentName: foundRow[2] || foundRow[0], totalXP: Number(foundRow[5]) || 0, unlockedLevels: { Math: Number(foundRow[6]) || 1, Science: Number(foundRow[7]) || 1, GK: Number(foundRow[8]) || 1 } } })).setMimeType(ContentService.MimeType.JSON); } // 3. RECOVER PIN if (data.action === "recover_pin") { var sName = (data.studentName || "").toLowerCase().trim(); var fName = (data.fatherName || "").toLowerCase().trim(); var dobTarget = (data.dob || "").toString().trim(); for (var k = 0; k < rows.length; k++) { var rowSName = rows[k][2].toString().toLowerCase().trim(); var rowFName = rows[k][3].toString().toLowerCase().trim(); var rowDOB = normalizeDate(rows[k][4]); if (rowSName === sName && rowFName === fName && (rowDOB === dobTarget || rowDOB.indexOf(dobTarget) !== -1)) { return ContentService.createTextOutput(JSON.stringify({ status: "success", username: rows[k][0], pin: rows[k][1] })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "❌ डेटा मैच नहीं हुआ! नाम, पिता का नाम और जन्मतिथि जांचें या WhatsApp पर पूछें।" })).setMimeType(ContentService.MimeType.JSON); } // 4. SUBMIT SCORE if (data.action === "submit_score") { var allSheet = ss.getSheetByName("All_Submissions") || ss.getSheets()[0]; allSheet.appendRow([ new Date(), data.username, data.subject, Number(data.score) || 0, Number(data.totalQuestions) || 5, Number(data.xpEarned) || 0 ]); for (var m = 0; m < rows.length; m++) { if (rows[m][0].toString().toLowerCase().trim() === (data.username || "").toLowerCase().trim()) { var rowNum = m + 2; var newXP = (Number(rows[m][5]) || 0) + (Number(data.xpEarned) || 0); uSheet.getRange(rowNum, 6).setValue(newXP); if (data.unlockedLevels) { uSheet.getRange(rowNum, 7).setValue(data.unlockedLevels.Math || 1); uSheet.getRange(rowNum, 8).setValue(data.unlockedLevels.Science || 1); uSheet.getRange(rowNum, 9).setValue(data.unlockedLevels.GK || 1); } uSheet.getRange(rowNum, 10).setValue(new Date()); break; } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } }
