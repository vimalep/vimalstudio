function parseQuizData(rawText) { // लाइन ब्रेक्स को नॉर्मलाइज़ करें const text = rawText.replace(/\r\n/g, '\n').replace(/\r/g, '\n'); // 1. सही उत्तर निकालना (A, B, C, या D को सटीक पकड़ेगा, चाहे कितने भी ** या स्पेस हों) const ansMatch = text.match(/(?:\*\*Answer:\*\*|Answer:|उत्तर:)\s*[\*]*\s*([A-D])(?:\)|\.|\s)/i); const correctAnswer = ansMatch ? ansMatch[1].toUpperCase() : 'B'; // 2. व्याख्या (Explanation) निकालना const expMatch = text.match(/(?:\*\*Explanation:\*\*|Explanation:|व्याख्या:)\s*([\s\S]*?)(?:\n---|$)/i); const explanation = expMatch ? expMatch[1].replace(/\*\*/g, '').trim() : ''; // 3. हैशटैग्स अलग निकालना const tagMatch = text.match(/(#[a-zA-Z0-9_\u0900-\u097F\s#]+)(?=\n\s*(?:\*\*Answer|\nAnswer|$))/); const hashtags = tagMatch ? tagMatch[1].trim() : ''; // 4. केवल Caption निकालना (यह Answer और Hashtags से पहले ही रुक जाएगा) let captionOnly = ''; const capMatch = text.match(/(?:\*\*Caption:\*\*|Caption:)\s*([\s\S]*?)(?=#|(?:\*\*Answer:\*\*|Answer:)|\n---|$)/i); if (capMatch) { captionOnly = capMatch[1].replace(/\*\*/g, '').trim(); } // 5. Facebook के लिए फाइनल क्लीन कैप्शन (बिना Answer/Explanation के) const cleanFacebookCaption = `${captionOnly}\n\n${hashtags}`.trim(); return { correctAnswer: correctAnswer, // जैसे: "B" explanation: explanation, // सिर्फ व्याख्या का टेक्स्ट facebookCaption: cleanFacebookCaption // सिर्फ कैप्शन + हैशटैग }; }
