async function generateQuestionsWithGemini() { let apiKey = document.getElementById('gemini-api-key').value.trim(); if (!apiKey) { apiKey = localStorage.getItem(GEMINI_KEY_STORAGE); } if (!apiKey) { alert("कृपया पहले API Key डालें!"); return; } const topic = document.getElementById('gemini-topic').value; const btn = document.getElementById('btn-ai-trigger'); btn.disabled = true; btn.innerHTML = `⏳ प्रश्न तैयार हो रहे हैं...`; const promptText = `Generate 5 competitive exam GK questions and answers in Hindi for topic: "${topic}". Output ONLY a raw JSON array of 5 objects with keys "q" and "a". No markdown, no ticks. Example: [{"q":"सिंधु घाटी सभ्यता का प्रमुख बंदरगाह कौन-सा था?","a":"लोथल"}]`; try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: promptText }] }] }) }); if (!response.ok) { const errData = await response.json(); throw new Error(errData.error?.message || `HTTP ${response.status}`); } const data = await response.json(); let reply = data.candidates?.[0]?.content?.parts?.[0]?.text || ""; reply = reply.replace(/```json/gi, '').replace(/```/g, '').trim(); const parsed = JSON.parse(reply); if (Array.isArray(parsed) && parsed.length > 0) { parsed.forEach(item => { if (item.q && item.a) { qaList.push({ q: item.q.trim(), a: item.a.trim() }); } }); autoSaveToStorage(); renderManageList(); alert(`🎉 5 नए प्रश्न सफलतापूर्वक लोड हो गए!`); } else { throw new Error("Invalid format"); } } catch (err) { alert("त्रुटि: " + err.message); console.error(err); } finally { btn.disabled = false; btn.innerHTML = `✨ Generate 5 Questions (AI)`; } }
